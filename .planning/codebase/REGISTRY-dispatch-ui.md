### Saga conventions

- One saga per file, named `<operation><Entity>Saga.ts` (e.g., `savePhaseDataSaga.ts`, `createPostSaga.ts`).
- Watchers use `takeLatest`, live alongside saga files.
- Try/catch dispatches failure action + (if user-facing) `enqueueSnackbar` via notification slice.

## HTTP & API Layer

- **Client:** `src/utils/axios.ts` — cookie-auth, 401 → refresh queue → retry, on failure: logout + `getNavigate()` redirect.

### Saga pattern
_Source: `src/features/carrier-portal/store/sagas/savePhaseDataSaga.ts:18-50`_

```typescript
import { call, put, select, takeLatest } from 'redux-saga/effects';
import type { PayloadAction } from '@reduxjs/toolkit';
import * as api from '../../../../utils/api/fleet/carrierPortalApi';
import { carrierPortalActions } from '../slices/carrierPortalSlice';

function* handleSaveCompany(action: PayloadAction<SaveCompanyRequest>): Generator {
  try {
    const token: string | null = yield* getToken();
    if (!token) {
      yield put(carrierPortalActions.saveCompanyFailure('No token available'));
      return;
    }
    yield call(api.saveCompany, token, action.payload);
    yield put(carrierPortalActions.saveCompanySuccess());
  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : 'Failed to save company';
    yield put(carrierPortalActions.saveCompanyFailure(message));
  }
}

export function* savePhaseDataSaga() {
  yield takeLatest(carrierPortalActions.saveCompanyRequest, handleSaveCompany);
  // ... per-phase workers
}
```
